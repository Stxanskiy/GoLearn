Импорт курса

поддерживается родной GoLearn-формат
Формат course.json 
{
  "slug": "my-course",            // латиница/цифры/дефис, уникален
  "title": "Мой курс",
  "category": "Kubernetes",       // цвет обложки, необязательно
  "difficulty": "advanced",
  "lessons": [
    {
      "slug": "intro", "title": "Введение",
      "kind": "theory",           // theory | quiz | lab | sim
      "format": "html",           // html | md
      "content": "<p>…</p>",
      "quiz": [ { "question": "2+2?", "options": ["3","4"], "correct": 2 } ],  // correct — 1-based
      "tasks": [ {
        "title": "Задание", "kind": "shell",
        "sandbox_image": "golearn/sandbox-k8s",
        "setup_script": "…", "check_script": "kubectl get …"   // exit 0 = решено
      } ]
    }
  ]
}
